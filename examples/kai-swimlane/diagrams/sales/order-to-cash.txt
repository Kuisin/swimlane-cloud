// 受付チャネルで入口が分かれる
phase (受付 | Intake) @intake #gray
  if (受注チャネルは？ | Order channel?) is (Web) than @channel #blue
    [system: 注文データ取込 | Import order] <external> +PO
      remark: 夜間バッチと即時取込の両方;
      remark.en: Both the nightly batch and on-demand import;
      id: import;
  else-if (電話 | Phone) than #orange
    [sales: 注文内容の聞き取り | Take the order by phone]
      desc: 品目・数量・納期を確認;
      desc.en: Confirm items, quantity and delivery date;
      remark: 記録形式は「品目 \| 数量 \| 納期」;
      remark.en: Record as "item \| quantity \| date";
      id: phone-order;
    [sales: 注文書（控え）を起票 | Draft the purchase order (copy)] +PO ~>
  else-if (その他 | Other) than #gray
    [sales: 受付不可を連絡 | Decline the order]
    [goto: closed]
  end-if
  [system: 注文を登録 | Register the order]
    label: 受注登録 | Order registration;
    id: registered;
end-phase

phase (与信と承認 | Credit and approval) @approval #teal
  [credit: 与信チェック | Credit check] +CR ..>
    desc: ```
    取引先マスタの与信枠と未回収残高を照合する。
    枠超過なら上長判断へ回す。
    ```;
    remark: 与信ポリシー第 3 条を参照;
    remark.en: See credit policy, article 3;
    id: credit-check;
  if (与信枠内？ | Within the credit limit?) is (はい | Yes) than @credit-ok #green
    [system: 与信 OK を記録 | Record credit OK]
  else-if (いいえ | No) than #red
    if (特別承認する？ | Special approval?) is (承認 | Approve) than @special #green
      [manager: 承認理由を記録 | Record the reason] +CR
    else-if (条件付き | With conditions) than #yellow
      [sales: 前払い条件を提示 | Offer prepayment terms]
      [sales: 顧客の回答を待つ | Wait for the customer] -.>
      // 回答が届いたら与信チェックからやり直す
      loop @credit-check
    else-if (却下 | Reject) than #red
      [sales: 受注を辞退 | Decline the order]
      [goto: closed]
    end-if
  end-if
  branch (監査記録 | Audit trail) #gray
    [system: 与信判断ログを保存 | Store the credit log] +CR
      remark: 保存期間は 7 年;
      remark.en: Kept for seven years;
  end-branch
  [manager: 受注を承認 | Approve the order]
    id: approved;
end-phase

fork (出荷 | Shipping) @fulfil #purple
  [warehouse: 在庫引当 | Allocate stock]
    id: allocate;
  if (在庫は十分？ | Enough stock?) is (はい | Yes) than #green
    [warehouse: ピッキングと梱包 | Pick and pack] => ./shipping-prep.txt
  else-if (いいえ | No) than #red
    [warehouse: 入荷待ちを登録 | Register a backorder]
    [sales: 納期変更を連絡 | Notify the new delivery date] -->
  end-if
  [warehouse: 出荷 | Ship]
    id: shipped;
case (請求 | Billing) #blue
  [billing: 請求書を発行 | Issue the invoice] +INV
    remark: 出荷日基準で計上;
    remark.en: Booked on the shipping date;
  [billing: 入金を待つ | Wait for payment]
case (通知 | Notification) #teal
  [system: 出荷通知メール | Shipping notice mail]
    id: notify;
  if (メール送信に失敗？ | Mail failed?) is (はい | Yes) than #red
    [sales: 電話で連絡 | Call the customer]
  else-if (いいえ | No) than #green
    [system: 送信済みを記録 | Record as sent]
  end-if
end-fork

section (クロージング | Closing) @closing #blue
  if (入金確認できた？ | Payment received?) is (はい | Yes) than @paid #green
    [billing: 消込を実行 | Reconcile the payment] +INV
  else-if (いいえ、期日超過 | No, overdue) than #orange
    [sales: 督促 | Send a reminder] ~>
    [sales: 回答を待つ | Wait for the reply]
    loop
  end-if
  [system: 受注をクローズ | Close the order] <terminal>
    id: closed;
end-section

[sales: 完了 | Done] <terminal>
  id: done;
  skip;

@end
